    function autosave() {
        const formData = new FormData(document.getElementById('shapeForm'));
        const data = {};
        formData.forEach((value, key) => {
            if (data[key]) {
                if (!Array.isArray(data[key])) data[key] = [data[key]];
                data[key].push(value);
            } else {
                data[key] = value;
            }
        });
        localStorage.setItem(STORAGE_KEY, JSON.stringify({
            timestamp: new Date().toLocaleString(),
            data: data
        }));
        document.getElementById('statusBar').innerText = `Last saved: ${new Date().toLocaleTimeString()}`;
    }

    function loadProgress() {
        const saved = localStorage.getItem(STORAGE_KEY);
        if (!saved) return;
        const { timestamp, data } = JSON.parse(saved);
        const form = document.getElementById('shapeForm');
        
        for (const [key, value] of Object.entries(data)) {
            const field = form.elements[key];
            if (!field) continue;
            if (field.type === 'radio') {
                const radio = Array.from(form.elements[key]).find(r => r.value === value);
                if (radio) radio.checked = true;
            } else if (field.type === 'checkbox') {
                field.checked = true;
            } else {
                field.value = value;
            }
        }
        document.getElementById('statusBar').innerText = `Restored progress from: ${timestamp}`;
    }

    function calculateResults() {

        if (!form.checkValidity()) {

            return;

            const val2 = parseInt(form.elements[`g${i+20}`]?.value || 0);
            const val3 = parseInt(form.elements[`g${i+40}`]?.value || 0);
            const val4 = parseInt(form.elements[`g${i+60}`]?.value || 0);
            scores[letter] = val1 + val2 + val3 + val4;
        }

        const ranked = Object.entries(scores)
            .map(([k, v]) => ({ letter: k, name: GIFT_LABELS[k], score: v }))
            .sort((a, b) => b.score - a.score);

        const top5 = ranked.slice(0, 5);

            </table>
